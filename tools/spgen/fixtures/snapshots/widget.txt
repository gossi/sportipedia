  """
  use Commanded.Projections.Ecto,
    application: Sportipedia.Catalog,
    repo: Sportipedia.Catalog.Repo,
    name: "equipment.widget_projection",
    schema_prefix: "catalog",
    consistency: :strong

  alias Sportipedia.Catalog.Equipment.Widget.Event.WidgetArchived

  alias Sportipedia.Catalog.Equipment.Widget.Event.WidgetEdited

  alias Sportipedia.Catalog.Equipment.Widget.WidgetInternal
  alias Sportipedia.Catalog.Equipment.Widget.WidgetReadModel

  @doc """
  Projects an WidgetCreated event to the widget read model.
  """
  project %WidgetCreated{} = event, _metadata, fn multi ->
    multi
    |> Ecto.Multi.insert(
      :widget_created,
      WidgetReadModel.insert_changeset(%WidgetReadModel{}, Map.from_struct(event))
    )
  end

  @doc """
  Projects an WidgetArchived event to hard-delete the widget read model.
  """
  project %WidgetArchived{} = event, _metadata, fn multi ->
    case WidgetInternal.widget_by_id(event.id) do
      nil ->
        multi

      %WidgetReadModel{} = widget ->
        multi
        |> Ecto.Multi.delete(:widget_archived, widget)
    end
  end

  @doc """

        multi
        |> Ecto.Multi.update(
          :widget_edited,
          WidgetReadModel.update_changeset(widget, attrs)

    end
  end

  @doc """
  Projects an WidgetRelabelled event to update the widget read model.

  project %WidgetRelabelled{} = event, _metadata, fn multi ->
    case WidgetInternal.widget_by_id(event.id) do
      nil ->

        multi
        |> Ecto.Multi.update(
          :widget_relabelled,

== services/api/lib/sportipedia/catalog/equipment/widget/public_api.ex ==
defmodule Sportipedia.Catalog.Equipment.Widget do
  @moduledoc """
  Public API for managing widgets in the sport.
  """
  alias Sportipedia.Architecture
  alias Sportipedia.Catalog
  alias Sportipedia.Catalog.Equipment.Widget.Command.ArchiveWidget
  alias Sportipedia.Catalog.Equipment.Widget.Command.CreateWidget
  alias Sportipedia.Catalog.Equipment.Widget.Command.EditWidget
  alias Sportipedia.Catalog.Equipment.Widget.Command.RelabelWidget
  alias Sportipedia.Catalog.Equipment.Widget.WidgetInternal
  alias Sportipedia.Catalog.Equipment.Widget.WidgetReadModel
  alias Sportipedia.Catalog.Repo
  alias Sportipedia.Support.ErrorClassifier

  alias Sportipedia.Support.JSONAPI.QueryBuilder

  alias Sportipedia.Support.Validators.UuidValidator

  @doc """
  Catalogs a new widget in the equipment catalog
  """
  @spec create_widget(%{
          required(:title) => String.t(),
          required(:slug) => String.t(),
          optional(:coachId) => String.t() | nil,
          optional(:description) => String.t() | nil,
          optional(:label) => String.t() | nil,
          optional(:teamId) => String.t() | nil
        }) :: Architecture.public_api(WidgetReadModel.t())
  def create_widget(params) do
    id = UUID.uuid4()
    cmd = CreateWidget.new(Map.put(params, "id", id))

    case Catalog.dispatch(cmd, consistency: :strong) do
      :ok ->
        {:ok, WidgetInternal.widget_by_id(id)}

      {:error, errors} ->
        ErrorClassifier.classify_error(errors)
    end
  end

  @doc """
  Archives a widget that is no longer used
  """
  @spec archive_widget(String.t()) :: Architecture.public_api()
  def archive_widget(id) do
    case Catalog.dispatch(ArchiveWidget.new(id: id), consistency: :strong) do
      :ok -> :ok
      {:error, errors} -> ErrorClassifier.classify_error(errors)
    end
  end

  @doc """
  Edits an existing widget
  """
  @spec edit_widget(%{
          required(:id) => String.t(),
          optional(:coachId) => String.t() | nil,
          optional(:description) => String.t() | nil,
          optional(:label) => String.t() | nil,
          optional(:slug) => String.t() | nil,
          optional(:teamId) => String.t() | nil,
          optional(:title) => String.t() | nil
        }) :: Architecture.public_api(WidgetReadModel.t())
  def edit_widget(params) do
    case Catalog.dispatch(EditWidget.new(params), consistency: :strong) do
      :ok ->
        {:ok, WidgetInternal.widget_by_id(params[:id] || params["id"])}

      {:error, errors} ->
        ErrorClassifier.classify_error(errors)
    end
  end

  @doc """
  List all widgets with filtering, sorting, and pagination
  """
  @spec list_widgets(JSONAPI.Config.t()) :: Architecture.public_api([WidgetReadModel.t()])
  def list_widgets(query) do
    {:ok, Repo.all(QueryBuilder.build(query, WidgetReadModel))}
  end

  @doc """
  Retrieve a single widget by its id or slug
  """
  @spec read_widget(String.t()) :: Architecture.public_api(WidgetReadModel.t())
  def read_widget(id_or_slug) do
    case lookup_widget(id_or_slug) do
      nil -> {:error, :not_found}
      read_model -> {:ok, read_model}
    end
  end

  defp lookup_widget(id_or_slug) do
    if UuidValidator.valid_uuid?(id_or_slug) do
      WidgetInternal.widget_by_id(id_or_slug)
    else
      WidgetInternal.widget_by_slug(id_or_slug)
    end
  end
end

== services/api/lib/sportipedia/catalog/equipment/widget/read_model.ex ==
defmodule Sportipedia.Catalog.Equipment.Widget.WidgetReadModel do
  @moduledoc """
  A cataloged widget, projected from the widget aggregate
  """
  use TypedEctoSchema
  import Ecto.Changeset
  @primary_key {:id, :binary_id, autogenerate: false}
  @timestamps_opts [type: :utc_datetime_usec]
  @schema_prefix "catalog"
  typed_schema "widget" do
    field :title, :string

    field :slug, :string
    field :archivedAt, :utc_datetime_usec
    field :coachId, :binary_id
    field :description, :string
    field :label, :string
    field :teamId, :binary_id
    timestamps()
  end

  @doc """
  Validates and prepares the widget read model for insertion.
  """
  def insert_changeset(read_model, attrs) do
    read_model
    |> cast(attrs, [:id, :title, :slug, :archivedAt, :coachId, :description, :label, :teamId])
    |> validate_required([:id, :title, :slug])
    |> unique_constraint(:slug, name: :widget_slug_index)
  end

  @doc """
  Validates and prepares the widget read model for update.
  """

== services/api/lib/sportipedia/catalog/equipment/widget/validators/unique_slug.ex ==
defmodule Sportipedia.Catalog.Equipment.Widget.Validators.UniqueSlug do
  @moduledoc """
  Validates that a widget slug is unique within the equipment.
  """
  use Vex.Validator
  alias Sportipedia.Catalog.Equipment.Widget.WidgetInternal

  @doc """
  Validates the given slug value.
  """
  @spec validate(String.t(), map()) :: :ok | {:error, term()}
  def validate(value, context) do
    with widget <- WidgetInternal.widget_by_slug(value),
         false <- is_nil(widget),
         false <- slug_belongs_to_widget?(widget, context) do
      {:error, :slug_exists}
    else
      _ -> :ok
    end
  end

  defp slug_belongs_to_widget?(widget, context) do
    id = Map.get(context, :id)

    id == widget.id
  end
end

== services/api/lib/sportipedia/catalog/equipment/widget/validators/widget_exists.ex ==
defmodule Sportipedia.Catalog.Equipment.Widget.Validators.WidgetExists do
  @moduledoc """
  Validates that a widget exists.

  The rule is asked of an address: only a value that can denote a
  widget is looked up. A malformed id addresses nothing, so no
  widget can hold it, and the answer is the one the pattern gives
  for an unknown id. An absent or blank id is not this rule's to answer —
  `presence` owns it, and the command's `validates` line keeps this validator
  off it.
  """
  use Vex.Validator
  alias Sportipedia.Catalog.Equipment.Widget.WidgetInternal
  alias Sportipedia.Support.Validators.UuidValidator

  @doc """
  Validates the given id value.
  """
  # `term()`, not `String.t()`: the value reaches here straight off the request,
  # and a request can name an id with anything at all.
  @spec validate(term(), map()) :: :ok | {:error, :not_found}
  def validate(id, _context) do
    # The existence check is a round trip on the primary key, and `Repo.get/2`
    # raises on a value it cannot put in that clause: nil, a blank string, an
    # id of the wrong shape. A raise here reaches the client as 500. A
    # value that cannot denote the resource denotes no resource — the pattern's
    # own verdict for an unknown id, given without leaving the process.
    if UuidValidator.valid_uuid?(id) do
      case WidgetInternal.widget_by_id(id) do
        nil -> {:error, :not_found}
        _widget -> :ok
      end
    else
      {:error, :not_found}
    end
  end
end
